OR

    { title: /^T/ },
    {
        $set: { "releaseDetails.0.rating": 8.0 }
    }
)

    firstName: "Cillian",

8. ==================== Delete all actors older than a particular age ====================
db.Actor.deleteMany({
    age: { $gt: 55 }
})

9. ==================== Update actor address where Actor ID is given ====================
db.Actor.updateOne(
    { actorId: 1 },
    {
        $set: {
            "address.street": "Sunset Boulevard",
            "address.city": "Los Angeles",
            "address.state": "California",
            "address.country": "USA",
            "address.pinCode": "90069"
        }
    }
)

Check - 
db.Actor.findOne({ actorId: 1 }).pretty()

10. ==================== Update the genre of a film directed by a particular director ====================
db.Film.updateOne(
    {
        directors: {
            $elemMatch: {
                firstName: "Christopher",
                lastName: "Nolan"
            }
        }
    },
    {
        $addToSet: {
            genre: "Mystery"
        }
    }
) 

OR 

db.Film.updateOne(
    {
        directors: {
            $elemMatch: {
                firstName: "Christopher",
                lastName: "Nolan"
            }
        }
    },
    {
        $set: {
            genre: ["Sci-Fi", "Mystery"]
        }
    }
)
