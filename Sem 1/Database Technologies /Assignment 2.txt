4. ==================== Update salary of all employees by Rs. 4000 ====================

db.Employee.updateMany(
    {},
    {
        $inc: { salary: 4000 }
    }
)

OR

db.Employee.updateMany(
    {},
    [
        {
            $set: {
                salary: { $add: ["$salary", 4000] }
            }
        }
    ]
)
