(Simple) Queries 

-------------------- (1) LIBRARY DATABASE --------------------

// (a) List all people who have issued the book "The Hobbit"
MATCH (p:Person)-[:IssuedBy]->(:Book {title:"The Hobbit"})
RETURN p.name;

// (b) Count the number of people who have read "The Hobbit"
MATCH (p:Person)-[:Read]->(:Book {title:"The Hobbit"})
RETURN count(p);

// (c) Add property "Number of books issued" for Mr. Joshi and set its value as the count
MATCH (p:Person {name:"Mr. Joshi"})-[:IssuedBy]->(b:Book)
WITH p, count(b) AS total
SET p.`Number of books issued` = total
RETURN p.name, p.`Number of books issued`;

// (d) List the names of publishers from Pune city
MATCH (p:Publisher {city:"Pune"})
RETURN p.name;


-------------------- (2) SONG DATABASE --------------------

// (a) List the names of songs written by "Kevin Parker"
MATCH (s:Song)-[:WrittenBy]->(:SongAuthor {name:"Kevin Parker"})
RETURN s.title;

// (b) List the names of record companies who have financed the song "Dracula"
MATCH (c:RecordingCompany)-[:Finances]->(:Song {title:"Dracula"})
RETURN c.name;

// (c) List the names of artists performing the song "Dracula"
MATCH (a:Artist)-[:Performs]->(:Song {title:"Dracula"})
RETURN a.name;

// (d) Name the songs recorded by the studio "Wave House"
MATCH (s:Song)-[:RecordedIn]->(:RecordingStudio {name:"Wave House"})
RETURN s.title;


-------------------- (3) EMPLOYEE DATABASE --------------------

// (a) List the names of employees in the department "Software Development"
MATCH (e:Employee)-[:Works_in]->(:Department {name:"Software Development"})
RETURN e.name;

// (b) List the projects along with their properties, controlled by department "Software Development"
MATCH (p:Project)-[:Controlled_by]->(:Department {name:"Software Development"})
RETURN p, properties(p);

// (c) List the departments along with the count of employees in it
MATCH (e:Employee)-[:Works_in]->(d:Department)
RETURN d.name, count(e);

// (d) List the skillset for an employee "Joel Guedes"
MATCH (:Employee {name:"Joel Guedes"})-[:Has_acquired]->(s:Skillset)
RETURN s.skill;


-------------------- (4) MOVIE DATABASE --------------------

// (a) Find all actors who have acted in a movie "Oppenheimer"
MATCH (a:Actor)-[:ACTED_IN]->(:Movie {title:"Oppenheimer"})
RETURN a.name;

// (b) Find all reviewer pairs, one following the other and both reviewing the same movie, and return entire subgraphs
MATCH p = (r1:Reviewer)-[:FOLLOWS]->(r2:Reviewer)-[:REVIEWED]->(m:Movie)<-[:REVIEWED]-(r1)
RETURN p;

// (c) Find all actors that acted in a movie together after 2010 and return the actor names and movie node
MATCH (a1:Actor)-[:ACTED_IN]->(m:Movie)<-[:ACTED_IN]-(a2:Actor)
WHERE m.releaseYear > 2010 AND a1.actorId < a2.actorId
RETURN a1.name, a2.name, m;

// (d) Find all movies produced by "Emma Thomas"
MATCH (:Producer {name:"Emma Thomas"})-[:PRODUCED]->(m:Movie)
RETURN m.title;


-------------------- (5) SOCIAL NETWORK DATABASE --------------------

// (a) Find all friends of "John", along with the year since when John knows them (no arrow: friendship can be stored in either direction)
MATCH (:Person {name:"John"})-[r:Friend_of]-(f:Person)
RETURN f.name, r.since;

// (b) List out the affiliations of John
MATCH (:Person {name:"John"})-[:Affiliated_to]->(a:Affiliation)
RETURN a.name;

// (c) Find all friends of John, who are born in the same year as John
MATCH (john:Person {name:"John"})-[:Friend_of]-(f:Person)
WHERE f.birthYear = john.birthYear
RETURN f.name;

// (d) List out the messages posted by John in his timeline, during the year 2015
MATCH (:Person {name:"John"})-[:Creates]->(:Timeline {year:2015})-[:Contains]->(m:Message)
RETURN m.content, m.sentOn;
