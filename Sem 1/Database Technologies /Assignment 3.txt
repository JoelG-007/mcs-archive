9. ==================== Find designation of employees who have made transaction of amount greater than Rs. 500. ====================


db.Transaction.aggregate([
  { $match: { "payment.amountPaid": { $gt: 500 } } },
  {
    $lookup: {
      from: "Employee",
      localField: "firstName",
      foreignField: "firstName",
      as: "employee"
    }
  },
  { $unwind: "$employee" },
  {
    $project: {
      _id: 0,
      designation: "$employee.designation"
    }
  }
]);


10. ==================== Find the total quantity of a particular item sold using Map Reduce. (Here "Laptop is used") ====================


db.Transaction.mapReduce(
  function() {
    if (this.transactionDet.itemName == "Laptop")
      emit(this.transactionDet.itemName, this.transactionDet.quantity);
  },
  function(key, values) {
    return Array.sum(values);
  }, 
  { out: "item_quantity" }
